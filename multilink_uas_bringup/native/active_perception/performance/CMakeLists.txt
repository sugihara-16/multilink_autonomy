cmake_minimum_required(VERSION 3.16)
project(uas_native_adapters LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc calib3d)
find_package(yaml-cpp REQUIRED)
add_executable(native_image_encoder native_image_encoder.cpp)
ament_target_dependencies(native_image_encoder rclcpp sensor_msgs geometry_msgs std_msgs)
target_link_libraries(native_image_encoder Eigen3::Eigen yaml-cpp ${OpenCV_LIBRARIES})
target_include_directories(native_image_encoder PRIVATE ${OpenCV_INCLUDE_DIRS})
find_package(rosgraph_msgs REQUIRED)
find_package(gz-transport13 REQUIRED)
find_package(gz-msgs10 REQUIRED)
add_executable(native_clock_bridge native_clock_bridge.cpp)
ament_target_dependencies(native_clock_bridge rclcpp rosgraph_msgs)
target_link_libraries(native_clock_bridge gz-transport13::core gz-msgs10::gz-msgs10)
